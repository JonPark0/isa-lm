# x86-64 build for MSVC and other CMake generators (GNU Make remains the
# primary build and the only one for ARM cross builds).
#   cmake -S . -B build/cmake -DML_ISA=avx2 [-DML_THREADS=ON]
cmake_minimum_required(VERSION 3.16)
project(legacy_ml C)

set(ML_ISA scalar CACHE STRING "scalar|sse2|sse41|avx2|avxvnni|avx512vnni")
option(ML_THREADS "pthread worker pool (not on MSVC)" OFF)

set(CHAIN_scalar scalar)
set(CHAIN_sse2 scalar sse2)
set(CHAIN_sse41 ${CHAIN_sse2} sse41)
set(CHAIN_avx2 ${CHAIN_sse41} avx2)
set(CHAIN_avxvnni ${CHAIN_avx2} avxvnni)
set(CHAIN_avx512vnni ${CHAIN_avx2} avx512vnni)
if(NOT DEFINED CHAIN_${ML_ISA})
  message(FATAL_ERROR "ML_ISA=${ML_ISA} not supported by CMake. Valid: scalar sse2 sse41 avx2 avxvnni avx512vnni")
endif()
set(CHAIN ${CHAIN_${ML_ISA}})

if(MSVC)
  set(FLAGS_sse2 "")
  set(FLAGS_sse41 "")
  set(FLAGS_avx2 /arch:AVX2)
  set(FLAGS_avxvnni /arch:AVX2)
  set(FLAGS_avx512vnni /arch:AVX512)
  set(BASE_OPTS /W3 /fp:precise)
  if(ML_THREADS)
    message(FATAL_ERROR "ML_THREADS needs pthreads; not available with MSVC")
  endif()
else()
  set(FLAGS_sse2 -msse2)
  set(FLAGS_sse41 -msse4.1)
  set(FLAGS_avx2 -mavx2 -mfma)
  set(FLAGS_avxvnni -mavx2 -mfma -mavxvnni)
  set(FLAGS_avx512vnni -mavx512f -mavx512bw -mavx512vnni -mfma)
  set(BASE_OPTS -Wall -Wextra -ffp-contract=off)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

string(TOUPPER ${ML_ISA} ISA_UP)
set(DEFS ML_ISA_${ISA_UP} _GNU_SOURCE)
foreach(i ${CHAIN})
  string(TOUPPER ${i} U)
  list(APPEND DEFS ML_ENABLE_${U})
endforeach()
if(ML_THREADS)
  list(APPEND DEFS ML_THREADS_PTHREAD)
  find_package(Threads REQUIRED)
endif()

file(GLOB LIB_SRC core/*.c ops/*.c model/*.c)
list(REMOVE_ITEM LIB_SRC ${CMAKE_SOURCE_DIR}/core/isa_check.c)
foreach(i ${CHAIN})
  file(GLOB K kernels/${i}/*.c)
  list(APPEND LIB_SRC ${K})
endforeach()

# Runs before the ISA check passes: no ISA flags.
add_library(ml_guard OBJECT core/isa_check.c app/main_guard.c)
target_include_directories(ml_guard PRIVATE ${CMAKE_SOURCE_DIR})
target_compile_definitions(ml_guard PRIVATE ${DEFS})
target_compile_options(ml_guard PRIVATE ${BASE_OPTS})

add_library(ml STATIC ${LIB_SRC})
target_include_directories(ml PUBLIC ${CMAKE_SOURCE_DIR})
target_compile_definitions(ml PUBLIC ${DEFS})
target_compile_options(ml PUBLIC ${BASE_OPTS} ${FLAGS_${ML_ISA}})
if(NOT MSVC)
  target_link_libraries(ml PUBLIC m)
endif()
if(ML_THREADS)
  target_link_libraries(ml PUBLIC Threads::Threads)
endif()

function(ml_exe name src)
  add_executable(${name} ${src} $<TARGET_OBJECTS:ml_guard>)
  target_link_libraries(${name} PRIVATE ml)
endfunction()

ml_exe(ml_run app/ml_run.c)
if(EXISTS ${CMAKE_SOURCE_DIR}/app/ml_bench.c)
  ml_exe(ml_bench app/ml_bench.c)
endif()

enable_testing()
foreach(t test_arena test_quant test_kernels test_sampler)
  ml_exe(${t} tests/${t}.c)
  add_test(NAME ${t} COMMAND ${t})
endforeach()
